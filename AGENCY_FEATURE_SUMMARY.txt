Core providers updated in lib/core/providers/providers.dart:

  final agenciesApiProvider = Provider<AgenciesApi>(
    (ref) => rab.getAgenciesApi(),
  );

  final agencyRemoteDataSourceProvider = Provider<AgencyRemoteDataSource>(
    (ref) => AgenciesRemoteDataSource(ref.read(agenciesApiProvider)),
  );

  final agencyRepositoryProvider = Provider<AgencyRepository>(
    (ref) => AgencyRepositoryImpl(ref.read(agencyRemoteDataSourceProvider)),
  );


💻 EXAMPLE USAGE
═══════════════════════════════════════════════════════════════════════════════

Load Agencies:
  final state = ref.watch(agenciesListProvider);
  await ref.read(agenciesListProvider.notifier).loadAgencies();

Create Agency:
  final useCase = ref.read(createAgencyUseCaseProvider);
  final agency = await useCase.call(name: 'Agency', description: 'Desc');

Get Agency Details:
  final state = ref.watch(agencyDetailProvider('agency-id'));
  await ref.read(agencyDetailProvider('agency-id').notifier).loadAgency();

Manage Staff:
  final staffState = ref.watch(agencyStaffListProvider('agency-id'));
  await ref.read(agencyStaffListProvider('agency-id').notifier).loadStaffs();
